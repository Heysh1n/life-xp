 * Решает проблему смены UUID на пиратских лаунчерах (Ely.by) и смены никнеймов на лицензии.

    /**
     * Алгоритм двойного маппинга (Dual-Key Identity Linking):
     * 1. Проверяем uuidLinker -> если есть, обновляем nameLinker новым ником (смена ника на лицензии).
     * 2. Проверяем nameLinker -> если есть, обновляем uuidLinker новым UUID (скачущий UUID на пиратке).
     * 3. Новый игрок -> генерируем Internal_Account_ID, связываем в обоих мапах и создаём CompoundTag.
     */
    public synchronized String resolveAccountId(UUID uuid, String rawName) {
        String name = rawName.toLowerCase(Locale.ROOT);
        String internalId;

        // ШАГ 1: Проверяем uuidLinker
        if (uuidLinker.containsKey(uuid)) {
            internalId = uuidLinker.get(uuid);
            nameLinker.put(name, internalId);
            setDirty();
        }
        // ШАГ 2: Проверяем nameLinker
        else if (nameLinker.containsKey(name)) {
            internalId = nameLinker.get(name);
            uuidLinker.put(uuid, internalId);
            setDirty();
        }
        // ШАГ 3: Новый игрок
        else {
            internalId = UUID.randomUUID().toString();
            uuidLinker.put(uuid, internalId);
            nameLinker.put(name, internalId);
            accounts.put(internalId, new CompoundTag());
            setDirty();
        }

        return internalId;
    }

    /**
     * Принудительная загрузка данных мода при входе игрока (игнорируя сброс ванильного NBT).
     */
    public synchronized void onPlayerJoin(ServerPlayer player) {
        UUID uuid = player.getUUID();
        String name = player.getScoreboardName();
        String internalId = resolveAccountId(uuid, name);

        CompoundTag accountData = accounts.computeIfAbsent(internalId, k -> new CompoundTag());

    /**
     * Сохранение данных мода для игрока в глобальную базу.
     */
    public synchronized void savePlayerData(ServerPlayer player) {
        UUID uuid = player.getUUID();
        String internalId = uuidLinker.get(uuid);
        if (internalId == null) {
            String name = player.getScoreboardName().toLowerCase(Locale.ROOT);
            internalId = nameLinker.get(name);
        }
        if (internalId == null) {
            internalId = resolveAccountId(uuid, player.getScoreboardName());
        }

        CompoundTag accountData = accounts.computeIfAbsent(internalId, k -> new CompoundTag());
        accountData.putLong("TotalXp", ExperienceUtils.getPlayerTotalXp(player));

        ListTag tagsTag = new ListTag();
//? if >= 26.1 {
        for (String tag : player.entityTags()) {
//?} else {
      /*for (String tag : player.getTags()) {*/
//?}
            if (tag.startsWith("lifexp_")) {
                tagsTag.add(StringTag.valueOf(tag));
            }
        }
        accountData.put("Tags", tagsTag);
        setDirty();
    }

    public synchronized void onPlayerQuit(ServerPlayer player) {

    }

    public synchronized void saveAll(MinecraftServer server) {

    public CompoundTag save(CompoundTag tag, HolderLookup.Provider registries) {
        CompoundTag accountsTag = new CompoundTag();
        for (Map.Entry<String, CompoundTag> entry : accounts.entrySet()) {
            accountsTag.put(entry.getKey(), entry.getValue().copy());
        }
        tag.put("Accounts", accountsTag);

        CompoundTag uuidTag = new CompoundTag();
        for (Map.Entry<UUID, String> entry : uuidLinker.entrySet()) {
            uuidTag.putString(entry.getKey().toString(), entry.getValue());
        }
        tag.put("UuidLinker", uuidTag);

        CompoundTag nameTag = new CompoundTag();
        for (Map.Entry<String, String> entry : nameLinker.entrySet()) {
            nameTag.putString(entry.getKey(), entry.getValue());
        }
        tag.put("NameLinker", nameTag);

        return tag;
    }

    public CompoundTag save(CompoundTag tag) {
        return save(tag, null);
    }

package com.hs1n.lifeXp_challenge.service;

import com.hs1n.lifeXp_challenge.LifeXpChallenge;

import net.minecraft.server.MinecraftServer;

public final class PresetManager {

    private PresetManager() {}

    /**
     * Вызывается в LifecycleEvent.SERVER_STARTED.
     * Проверяет, является ли сервер выделенным (Dedicated).
     * Dedicated -> форсируем хардкорный пресет (Core).
     * Integrated (одиночная игра) -> щадящий пресет (Vanilla+).
     */
    public static void onServerStarted(MinecraftServer server) {
        if (!LifeXpConfig.INSTANCE.isPresetInitialized()) {
            if (server.isDedicatedServer()) {
                LifeXpPresets.applyCore();
                LifeXpChallenge.LOGGER.info("[LIFE-XP] Dedicated server detected. Auto-configured HARDCORE (Core) preset.");
            } else {
                LifeXpPresets.applyVanillaPlus();
                LifeXpChallenge.LOGGER.info("[LIFE-XP] Integrated server detected. Auto-configured LENIENT (Vanilla+) preset.");
            }
            LifeXpConfig.INSTANCE.setPresetInitialized(true);
            LifeXpConfig.save();
        }
    }

    /**
     * Проверка прав оператора (Уровень 2).
     */
//? if >= 26.1 {

     * Проверка наличия тега выбора пресета у игрока.

    public static boolean hasPresetTag(ServerPlayer player) {
//? if >= 26.1 {
        return player.entityTags().contains(LifeXpCommand.PRESET_TAG);
//?} else {
      /*return player.getTags().contains(LifeXpCommand.PRESET_TAG);*/

    /**
     * Безопасное открытие меню выбора пресета при первом входе.
     * Открывается ТОЛЬКО если игрок имеет права оператора (hasPermissions(2)).
     * Обычные игроки не имеют доступа и сразу помечаются тегом, предотвращая открытие меню и эксплойт.
     */
    public static void checkAndPromptPreset(ServerPlayer player) {
        if (!LifeXpConfig.INSTANCE.isPresetInitialized() && !hasPresetTag(player)) {
            if (hasOperatorPermissions(player)) {
                LifeXpNetworking.sendOpenPresetScreen(player);
            } else {
                player.addTag(LifeXpCommand.PRESET_TAG);
            }
        }
    }

    public static void init() {

        BlockEvent.BREAK.register(XpHarvesterService::onBlockBreak);
//?} else {
      /*BlockEvent.BREAK.register((level, pos, state, player, xp) -> onBlockBreak(level, pos, state, player));*/

        BlockEvent.PLACE.register(XpHarvesterService::onBlockPlace);

    public static EventResult onBlockPlace(Level level, BlockPos pos, BlockState state, @Nullable Entity entity) {
        if (level instanceof ServerLevel serverLevel) {
            if (state.is(Blocks.ANCIENT_DEBRIS)) {
                PlacedBlockTracker tracker = PlacedBlockTracker.get(serverLevel);
                tracker.add(pos);
                tracker.setDirty();
            }
        }
        return EventResult.pass();
    }

    public static int getHarvesterLevel(ItemStack tool) {
        if (tool == null || tool.isEmpty()) return 0;
        ItemEnchantments enchantments = tool.getOrDefault(DataComponents.ENCHANTMENTS, ItemEnchantments.EMPTY);
        for (Holder<Enchantment> holder : enchantments.keySet()) {
            if (holder.is(ENCHANTMENT_ID)) {
                return enchantments.getLevel(holder);
            }
        }
        return 0;
    }

    public static EventResult onBlockBreak(Level level, BlockPos pos, BlockState state, ServerPlayer player) {
        if (!(level instanceof ServerLevel serverLevel) || player == null || player.isCreative()) {
            return EventResult.pass();
        }
        // Skip if vein mining is in progress (prevents recursive triggers)
        if (OreVeinMinerService.IS_VEIN_MINING.get()) {
            return EventResult.pass();
        }

        ItemStack tool = player.getMainHandItem();
        int enchantLevel = getHarvesterLevel(tool);
        if (enchantLevel <= 0) {
            return EventResult.pass();
        }

        RandomSource random = serverLevel.getRandom();

        // 1. Созревший CropBlock (isMaxAge) -> спавн 1-2 ExperienceOrb.
        if (state.getBlock() instanceof CropBlock crop && crop.isMaxAge(state)) {
            int totalCropXp = 1 + random.nextInt(2);
            ExperienceOrbHelper.spawnClumpedOrb(serverLevel, Vec3.atCenterOf(pos), totalCropXp);
            return EventResult.pass();
        }

        // 2. Ancient Debris -> 1% от общего опыта игрока, жесткий хардкап Math.min(bonus, 500).
        if (state.is(Blocks.ANCIENT_DEBRIS)) {
            PlacedBlockTracker tracker = PlacedBlockTracker.get(serverLevel);
            if (tracker.contains(pos)) {
                tracker.remove(pos);
                tracker.setDirty();
                return EventResult.pass();
            }
            long totalPlayerXp = ExperienceUtils.getPlayerTotalXp(player);
            int bonus = (int) Math.floor(totalPlayerXp * 0.01);
            bonus = Math.min(bonus, 500);
            if (bonus > 0) {
                spawnOreXp(serverLevel, Vec3.atCenterOf(pos), bonus);
            }
            return EventResult.pass();
        }

        // 3. Ванильные руды (железо, медь, золото, лазурит, редстоун, алмазы) -> бонусный скейлинг опыта от уровня чар.
        int bonusXp = calculateOreBonus(state, enchantLevel, random);
        if (bonusXp > 0) {
            spawnOreXp(serverLevel, Vec3.atCenterOf(pos), bonusXp);
            AdvancementService.addOreXp(player, bonusXp);
        }

        // 4. Ore Vein Miner (XP Harvester level >= 2 on pickaxe)
        OreVeinMinerService.tryVeinMine(serverLevel, player, pos, state, tool, enchantLevel);

        return EventResult.pass();
    }

    private static int calculateOreBonus(BlockState state, int level, RandomSource random) {
        return switch (state) {
            case BlockState s when s.is(BlockTags.IRON_ORES) || s.is(BlockTags.COPPER_ORES) -> (1 + random.nextInt(2)) * level;
            case BlockState s when s.is(BlockTags.GOLD_ORES) -> (2 + random.nextInt(2)) * level;
            case BlockState s when s.is(LAPIS_ORES) -> (2 + random.nextInt(4)) * level;
            case BlockState s when s.is(REDSTONE_ORES) -> (1 + random.nextInt(5)) * level;
            case BlockState s when s.is(DIAMOND_ORES) -> (3 + random.nextInt(5)) * level;
            default -> 0;
        };
    }

    private static void spawnOreXp(ServerLevel level, Vec3 pos, int amount) {
        try {
            ExperienceSourceHelper.IS_ORE_XP.set(true);
            ExperienceOrbHelper.spawnClumpedOrb(level, pos, amount);
        } finally {
            ExperienceSourceHelper.IS_ORE_XP.set(false);
        }
    }

    private XpHarvesterService() {}
}
